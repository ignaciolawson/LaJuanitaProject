// Generado a partir del esquema real, migraciones V1..V6.
//
// Al 2026-08-14 el diagrama está AL DÍA: comparado tabla por tabla y columna
// por columna contra el catálogo de una base con las seis migraciones
// aplicadas, coinciden las 22 tablas y todas sus columnas. Incluye lo que
// agregaron V4 (`nombre` + `apellido` en vez de `nombre_completo`), V5
// (`usuario.debe_cambiar_password`) y V6 (la columna generada
// `bloqueo_sala.periodo`).
//
// La cabecera anterior decía lo contrario -- "NO REFLEJA V3..V6" -- y era
// falso: invitaba a "corregir" el diagrama volviendo a `nombre_completo`, que
// es justo el error que quería prevenir.
//
// Lo que el diagrama sigue sin poder mostrar son las constraints y los
// triggers, y eso es harina de otro costal: está explicado abajo. V6 sumó
// 12 CHECK, 3 triggers, un EXCLUDE en `bloqueo_sala` y un UNIQUE parcial en
// `artista.id_usuario`, todos por hallazgos de la auditoría adversarial
// (`docs/db/auditoria-2026-08-12.md`), y ninguno se ve acá.
//
// Para ver el diagrama: entrar a https://dbdiagram.io > New Diagram >
// borrar el ejemplo > pegar este archivo completo.
//
// ----------------------------------------------------------------------------
// LO QUE ESTE DIAGRAMA **NO** PUEDE MOSTRAR
//
// dbdiagram dibuja tablas y relaciones, pero no constraints de exclusión ni
// triggers -- y ahí es donde vive buena parte de la lógica de este sistema.
// Las reglas que la base hace cumplir por su cuenta son:
//
//   1. Nunca dos reservas solapadas en la misma sala (constraint EXCLUDE, la
//      única forma de que aguante dos personas reservando en el mismo instante).
//   2. Una sala bloqueada no acepta reservas, y no se bloquea una sala con
//      reservas activas adentro (triggers, con lock de fila para que no se
//      cuelen por concurrencia).
//   3. Solo se puede hacer en cada sala lo que la matriz sala_tipo_uso permite.
//   4. La inscripción que se descuenta tiene que ser del alumno que asiste.
//   5. El premaster de un trabajo de mastering no se libera sin pago
//      registrado, salvo excepción explícita y con motivo.
//   6. Los estados de trabajo_mastering y de release solo avanzan.
//   7. Una nota de profesor va sobre una clase de ese mismo alumno.
//   8. (V6) Dos bloqueos no se solapan sobre la misma sala.
//   9. (V6) El pago que respalda un premaster liberado no se puede anular ni
//      borrar sin marcar antes el trabajo como liberado_sin_pago.
//  10. (V6) No se borran filas de `pago` ni de `trabajo_mastering`: se anulan.
//
// El detalle y el porqué de cada una está comentado en V1__baseline.sql y en
// V6__integridad_auditoria.sql.
//
// OJO con la regla 2: solo se garantiza bajo READ COMMITTED. En REPEATABLE
// READ y en SERIALIZABLE el trigger lee un snapshot viejo y la reserva entraría
// igual -- está medido. Desde V6 la transacción se rechaza en vez de mentir.
//
// ----------------------------------------------------------------------------
// CAMBIOS RESPECTO DE LA v2 (el diagrama que estaba antes acá)
//
//   · NUEVA `inscripcion`: acá vive "Juan compró el curso de DJ inicial, 8
//     clases, $X, con Tomás". Sin ella no se podía saber cuántas clases le
//     quedan a nadie, que es lo que el relevamiento marca como faltante hoy.
//   · NUEVAS `tipo_uso` + `sala_tipo_uso`: la matriz de qué se puede hacer en
//     cada sala. Va como datos y no como código para que cambiarla no requiera
//     tocar el sistema.
//   · `historial_clase` SE ELIMINÓ. La reemplaza `reserva_participante`: una
//     clase puede ser grupal, así que la asistencia es por participante.
//   · `usuario.rol` pasa a cuatro niveles de permiso (ADMIN / DIRECTIVO /
//     STAFF / USUARIO), separados de las relaciones de negocio.
//   · `alumno.disciplina` y `alumno.nivel_actual` se mudaron a `inscripcion`.
//   · `reserva` gana `id_reserva_recupera`: ninguna clase se pierde nunca.
//   · No se modela `sede`: Córdoba salió del alcance el 2026-08-11.

Table usuario {
  id_usuario        bigint      [pk, increment]
  nombre            varchar(80)  [not null]
  apellido          varchar(80)  [not null, note: 'Separado del nombre (V4): el listado se ordena y filtra por apellido']
  email             varchar(150) [not null, unique]
  telefono          varchar(40)  [unique]
  password_hash     varchar(255) [not null]
  rol               varchar(20)  [not null, default: 'USUARIO', note: 'ADMIN / DIRECTIVO / STAFF / USUARIO']
  foto_perfil       varchar(500)
  estado_presencia  varchar(20)  [not null, default: 'ACTIVO', note: 'ACTIVO / AUSENTE / OCUPADO / EN_CLASE']
  bio               text
  especializacion   varchar(150)
  activo            boolean      [not null, default: true]
  debe_cambiar_password boolean  [not null, default: false, note: 'TRUE si la contraseña la generó administración y todavía no la cambió (V5)']
  fecha_creacion    timestamptz  [not null]
  ultimo_acceso     timestamptz

  Note: '''
  Identidad única de login. Cualquier persona -- alumno, profesor o cliente
  ocasional -- es primero un usuario.

  `rol` es el eje de PERMISOS (qué puede administrar del sistema) y es
  independiente de las relaciones de negocio (tener fila en alumno o profesor),
  que son las que arman el menú del portal. Por eso Ghezz puede ser STAFF *y*
  profesor *y* alquilar una cabina para él, sin contradicción.

  El email es único sin distinguir mayúsculas. Nadie se borra: se desactiva.
  '''
}

Table sala {
  id_sala      bigint       [pk, increment]
  nombre_sala  varchar(100) [not null, unique]
  descripcion  text
  activa       boolean      [not null, default: true]
  orden        smallint     [not null, default: 0, note: 'Orden en el calendario']

  Note: '''
  Tres salas, y las tres son "cabinas" (lugares donde se toca):
  Sala 1, Sala 2 y Cabina de grabación.

  Las dos primeras son de práctica y sirven para todo. La de grabación tiene
  equipos y cámara pero NO tiene silla, tele ni escritorio, así que solo sirve
  para tocar y grabarse. Qué se puede hacer en cada una está en sala_tipo_uso.
  '''
}

Table tipo_uso {
  id_tipo_uso  bigint       [pk, increment]
  codigo       varchar(40)  [not null, unique]
  nombre       varchar(100) [not null]
  es_clase     boolean      [not null, default: false]
  color        varchar(20)  [note: 'Color en el calendario']
  activo       boolean      [not null, default: true]

  Note: 'CLASE_DJ, PRODUCCION_MUSICAL, MENTORIA, MIX_MASTERING, ALQUILER_CABINA, GRABACION_SET.'
}

Table sala_tipo_uso {
  id_sala      bigint [ref: > sala.id_sala, not null]
  id_tipo_uso  bigint [ref: > tipo_uso.id_tipo_uso, not null]
  advertencia  text   [note: 'Permitido, pero con aviso al reservar']

  Indexes {
    (id_sala, id_tipo_uso) [pk]
  }

  Note: '''
  LA MATRIZ de qué se puede hacer en cada sala.

                       | Sala 1 | Sala 2 | Cabina grabación
   Clase de DJ         |   si   |   si   |  si (con aviso)
   Producción musical  |   si   |   si   |       no
   Mentoría            |   si   |   si   |       no
   Mix & Mastering     |   si   |   si   |       no
   Alquiler de cabina  |   si   |   si   |       no
   Grabación de set    |   no   |   no   |       si

  Es una tabla y no reglas escritas en el código a propósito: el día que
  compren una tele y una silla para la cabina de grabación, la regla se cambia
  desde una pantalla, sin migración ni deploy.

  `reserva` tiene una clave foránea COMPUESTA contra esta tabla, así que una
  combinación no permitida es imposible de guardar.
  '''
}

Table inscripcion {
  id_inscripcion      bigint        [pk, increment]
  id_alumno           bigint        [ref: > alumno.id_alumno, not null]
  id_profesor         bigint        [ref: > profesor.id_profesor, note: 'Profesor a cargo — asignación explícita']
  disciplina          varchar(20)   [not null, note: 'DJ / PRODUCCION / MENTORIA']
  nivel               varchar(20)   [note: 'INICIAL / INTERMEDIO / AVANZADO']
  clases_contratadas  smallint      [not null, note: 'DJ = 8, Producción = 16']
  precio_total        decimal(14,2) [not null]
  moneda              varchar(3)    [not null, default: 'ARS', note: 'ARS / USD']
  cotizacion_dolar    decimal(14,4) [note: 'Obligatoria si moneda = USD']
  fecha_inicio        date
  estado              varchar(20)   [not null, default: 'ACTIVA', note: 'ACTIVA / COMPLETADA / CANCELADA / PAUSADA']
  notas               text
  fecha_creacion      timestamptz   [not null]

  Note: '''
  La pieza que faltaba en el modelo original: acá vive "Juan compró el curso de
  DJ inicial, 8 clases, $X, con Tomás".

  El curso es CERRADO (8 clases), no un paquete de clases sueltas, y ninguna
  clase se pierde nunca: si falta el alumno o falta el profesor, se recupera en
  otra fecha. Por eso no hay fecha de fin comprometida -- el curso termina
  cuando se dictaron las clases contratadas.

  El profesor se asigna ACÁ y no en el alumno, así el mismo alumno puede tener
  un profe para DJ y otro para mentoría. "Mis Alumnos" del portal sale de esta
  tabla, no de las reservas: que otro profe cubra una clase suelta no le
  transfiere el alumno.

  REGLA: una sola inscripción ACTIVA por disciplina. Se puede hacer DJ inicial
  y mentoría a la vez; no DJ inicial y DJ avanzado. (Índice único parcial.)
  '''
}

Table reserva {
  id_reserva            bigint      [pk, increment]
  id_sala               bigint      [ref: > sala.id_sala, not null]
  id_tipo_uso           bigint      [ref: > tipo_uso.id_tipo_uso, not null]
  id_profesor           bigint      [ref: > profesor.id_profesor]
  fecha                 date        [not null]
  hora_inicio           time        [not null]
  hora_fin              time        [not null]
  estado                varchar(20) [not null, default: 'CONFIRMADA', note: 'CONFIRMADA / MODIFICADA / CANCELADA / REPROGRAMADA / FINALIZADA']
  notas                 text
  id_reserva_recupera   bigint      [ref: > reserva.id_reserva, note: 'A qué clase reemplaza esta']
  motivo_reprogramacion text
  id_usuario_creo       bigint      [ref: > usuario.id_usuario]
  id_usuario_modifico   bigint      [ref: > usuario.id_usuario]
  fecha_creacion        timestamptz [not null]
  fecha_modificacion    timestamptz
  periodo               tsrange     [note: 'Columna generada: el rango ocupado']

  Note: '''
  Un bloque de tiempo en una sala. Puede ser una clase, un alquiler, una sesión
  de mastering o una grabación.

  Quiénes participan NO está acá sino en reserva_participante, porque una clase
  puede ser grupal y cada alumno viene de su propia inscripción.

  UNA RESERVA OCUPA SU FRANJA salvo que esté CANCELADA o REPROGRAMADA. Esa
  definición es única y la usan los tres mecanismos que la necesitan.

  La regla más importante del sistema -- nunca dos reservas solapadas en la
  misma sala -- va como constraint EXCLUDE y no como validación en el código,
  porque es la única forma de que aguante a dos personas reservando en el mismo
  instante.
  '''
}

Table reserva_participante {
  id_participacion  bigint      [pk, increment]
  id_reserva        bigint      [ref: > reserva.id_reserva, not null]

  estado_asistencia varchar(20) [not null, default: 'PENDIENTE', note: 'PENDIENTE / PRESENTE / AUSENTE / AUSENTE_JUSTIFICADO / CANCELADA']
  observaciones     text
  fecha_registro    timestamptz [not null]

  Note: '''
  Reemplaza a `historial_clase` del modelo viejo. Como una clase puede ser
  grupal, la asistencia es por participante y no por reserva.

  `id_inscripcion` es lo que descuenta la clase del curso contratado. Un

  eso se le podía descontar la clase de uno al paquete de otro.

}

Table pago {
  id_pago              bigint        [pk, increment]
  id_usuario           bigint        [ref: > usuario.id_usuario, not null, note: 'Quién paga']
  id_usuario_registra  bigint        [ref: > usuario.id_usuario]

  cotizacion_dolar     decimal(14,4) [note: 'Obligatoria si moneda = USD']
  medio_pago           varchar(30)   [not null, note: 'EFECTIVO / TRANSFERENCIA / PAYPAL / CUENTA_EEUU / OTRO']

  comprobante_invalido boolean       [not null, default: false]
  fecha_pago           date          [not null]
  fecha_registro       timestamptz   [not null]

  Note: '''
  Un pago salda UNA cosa, exactamente una: una inscripción, una reserva, un
  trabajo de mastering o una venta de equipo. Si alguien paga dos servicios
  juntos, son dos filas -- si no, el mismo monto se cuenta dos veces en los
  reportes por línea de negocio.

  `monto` es lo efectivamente cobrado y `descuento_porcentaje` es el porcentaje
  aplicado, así la caja es la suma de `monto` sin recalcular nada. Todo
  descuento exige justificación escrita.

  Los comprobantes no se borran: se marcan como inválidos.
  '''
}

Table trabajo_mastering {
  id_trabajo               bigint        [pk, increment]
  id_cliente_usuario       bigint        [ref: > usuario.id_usuario, note: 'Vacío si el cliente no tiene cuenta']
  nombre_cliente_externo   varchar(150)
  contacto_cliente_externo varchar(150)
  id_profesor_asignado     bigint        [ref: > profesor.id_profesor]
  tipo_trabajo             varchar(20)   [not null, note: 'MIX / MASTER / MIX_MASTER']
  nombre_track             varchar(200)  [not null]
  precio_acordado          decimal(14,2)
  moneda                   varchar(3)    [not null, default: 'USD']
  cotizacion_dolar         decimal(14,4)
  revisiones_incluidas     smallint      [not null, default: 3]
  revisiones_realizadas    smallint      [not null, default: 0]
  fecha_estimada           date
  fecha_entrega_real       date
  estado                   varchar(20)   [not null, default: 'A_CONFIRMAR', note: 'A_CONFIRMAR / EN_PROCESO / ENTREGADO / PAGADO / DEBE / CANCELADO']
  url_material_cliente     varchar(500)  [note: 'Lo que mandó el cliente']
  url_master               varchar(500)  [note: 'Se entrega antes de cobrar']
  url_premaster            varchar(500)  [note: 'Se retiene hasta el pago']
  premaster_liberado       boolean       [not null, default: false]
  liberado_sin_pago        boolean       [not null, default: false]
  motivo_liberacion        text
  id_usuario_libera        bigint        [ref: > usuario.id_usuario]
  notas_internas           text
  fecha_creacion           timestamptz   [not null]

  Note: '''
  El único servicio que puede quedar en debe: Ghezz entrega primero y cobra
  después. Entrega el MASTER; retiene el PREMASTER (el que el cliente necesita
  para las discográficas) hasta que el pago esté registrado.

  El sistema NO guarda los audios -- un WAV pesa 80-100 MB y un set en video
  entre 2 y 4 GB. Guarda el LINK y controla cuándo el cliente puede verlo, que
  para el caso es lo mismo y no obliga a nadie a subir 100 MB por una web.

  Un trigger impide liberar el premaster sin pago. La excepción existe a
  propósito, porque Ghezz cobra con flexibilidad según el cliente: se puede
  liberar igual, pero queda registrado quién lo hizo y por qué.

  A diferencia del resto de las tablas, acá NO se exige cotización para los
  montos en USD: el precio acordado es del presupuesto y la cotización que
  importa es la del día del cobro, que vive en `pago`.
  '''
}

Table artista {
  id_artista        bigint       [pk, increment]
  id_usuario        bigint       [ref: > usuario.id_usuario, note: 'Los artistas hoy no tienen login; queda preparado']
  nombre_artistico  varchar(150) [not null]
  nombre_real       varchar(150)
  email_contacto    varchar(150)
  telefono          varchar(40)
  instagram         varchar(100)
  confirmado        boolean      [not null, default: false]
  bio               text
  fecha_alta        timestamptz  [not null]
}

Table release {
  id_release      bigint       [pk, increment]
  codigo_release  varchar(20)  [not null, unique, note: 'ID propio del sello: LJ020, LJ021...']
  id_artista      bigint       [ref: > artista.id_artista, not null]
  nombre_release  varchar(200) [not null]
  tipo_release    varchar(20)  [note: 'SINGLE / EP / REMIX / ALBUM']
  genero          varchar(80)
  portada_path    varchar(500)
  fecha_estimada  date
  fecha_real      date
  estado          varchar(20)  [not null, default: 'A_CONFIRMAR', note: 'A_CONFIRMAR / CONFIRMADO / EN_DISTRIBUCION / PUBLICADO']
  sistema_promo   boolean      [not null, default: false, note: 'Enviado a la red internacional de DJs']
  notas           text
  fecha_creacion  timestamptz  [not null]

  Note: 'El código es único y correlativo. El estado solo avanza, nunca retrocede (trigger).'
}

Table contrato_sello {
  id_contrato    bigint       [pk, increment]
  id_artista     bigint       [ref: > artista.id_artista, not null]
  id_release     bigint       [ref: > release.id_release, note: 'Vacío si el contrato cubre al artista en general']
  archivo_path   varchar(500) [not null]
  fecha_firma    date
  observaciones  text
  fecha_carga    timestamptz  [not null]
}

Table notificacion {

  id_usuario_destino bigint      [ref: > usuario.id_usuario, not null]

  contenido          text        [not null]
